PostgreSQL - Two overlapping transactions

--- Session A ---
BEGIN;
UPDATE products SET price = 26000 WHERE product_id = 1;
-- Row locked, not yet committed

--- Session B (at same time) ---
BEGIN;
UPDATE products SET price = 27000 WHERE product_id = 1;
-- Session B WAITS - blocked by Session A lock

--- Session A ---
COMMIT;
-- Lock released

--- Session B ---
-- Now unblocked, proceeds
COMMIT;

Result: Session B waited for Session A. No lost update. Locking works as expected.
